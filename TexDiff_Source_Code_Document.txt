import os
import torch
import torch.nn as nn
from PIL import Image
from diffusers import AutoPipelineForText2Image

class TexDiffPipeline:
    """
    Enterprise Core Diffusion Engine for Seamless Textile CAD Pattern Synthesis.
    Enforces Conv2d circular padding on U-Net and VAE for true periodic boundary condition repeats.
    Features self-healing scheduler state to prevent IndexError step desynchronization.
    """
    def __init__(self, model_id: str = "stabilityai/sd-turbo"):
        self.model_id = model_id
        self._init_device_and_precision()
        self._load_pipeline()

        # Strict Defect & Zero-Human Guardrails
        self.default_negative_prompt = (
            "human, person, woman, girl, model, mannequin, face, eyes, body, limbs, fingers, "
            "wearing clothes, dress, saree photo, garment mockup, 3D perspective, depth of field, "
            "blurry edges, seam cut, photo background, shadows, folded fabric, table, room interior, "
            "photorealistic wrinkles, distorted proportions, low quality"
        )

    def _init_device_and_precision(self):
        """Enforces fp16 on CUDA/MPS with safe fallback to float32 on CPU."""
        try:
            if torch.cuda.is_available():
                self.device = "cuda"
                self.dtype = torch.float16
                self.variant = "fp16"
            elif hasattr(torch.backends, "mps") and torch.backends.mps.is_available():
                self.device = "mps"
                self.dtype = torch.float16
                self.variant = None
            else:
                self.device = "cpu"
                self.dtype = torch.float32
                self.variant = None
        except Exception as e:
            print(f"[TexDiff] Hardware check warning: {e}. Defaulting to CPU float32.")
            self.device = "cpu"
            self.dtype = torch.float32
            self.variant = None

        print(f"[TexDiff] Diffusion hardware selected: {self.device} | Precision: {self.dtype}")

    def _load_pipeline(self):
        """Loads AutoPipelineForText2Image with recovery handling."""
        try:
            kwargs = {"torch_dtype": self.dtype}
            if self.variant:
                kwargs["variant"] = self.variant

            self.pipe = AutoPipelineForText2Image.from_pretrained(
                self.model_id,
                **kwargs
            )
            self.pipe.to(self.device)
            print(f"[TexDiff] Pipeline successfully loaded on {self.device}.")
        except Exception as e:
            if self.device != "cpu":
                print(f"[TexDiff] CUDA init failed ({e}). Falling back to CPU float32...")
                self.device = "cpu"
                self.dtype = torch.float32
                self.variant = None
                self.pipe = AutoPipelineForText2Image.from_pretrained(
                    self.model_id,
                    torch_dtype=torch.float32
                )
                self.pipe.to("cpu")
            else:
                raise RuntimeError(f"[TexDiff] Fatal error loading diffusion pipeline: {e}")

    def set_circular_padding(self, enable: bool = True):
        """
        Dynamically configures padding_mode='circular' across all Conv2d layers in U-Net and VAE.
        This wraps latent and pixel boundaries periodically, eliminating seam artifacts.
        """
        mode = "circular" if enable else "zeros"
        try:
            if hasattr(self.pipe, "unet") and self.pipe.unet is not None:
                for module in self.pipe.unet.modules():
                    if isinstance(module, nn.Conv2d):
                        module.padding_mode = mode

            if hasattr(self.pipe, "vae") and self.pipe.vae is not None:
                for module in self.pipe.vae.modules():
                    if isinstance(module, nn.Conv2d):
                        module.padding_mode = mode
        except Exception as e:
            print(f"[TexDiff] Warning setting circular padding: {e}")

    def _reset_scheduler(self):
        """
        Self-healing scheduler state reset.
        Guarantees that `_step_index` and timesteps are completely pristine before each run,
        preventing `IndexError: index X is out of bounds for dimension 0 with size Y`.
        """
        try:
            if hasattr(self.pipe, "scheduler") and self.pipe.scheduler is not None:
                if hasattr(self.pipe.scheduler, "_step_index"):
                    self.pipe.scheduler._step_index = None
                if hasattr(self.pipe.scheduler, "is_scale_input_called"):
                    self.pipe.scheduler.is_scale_input_called = False
                if hasattr(self.pipe.scheduler, "step_index"):
                    try:
                        self.pipe.scheduler.step_index = None
                    except Exception:
                        pass
        except Exception as e:
            print(f"[TexDiff] Scheduler reset notice: {e}")

    def _recreate_scheduler_fresh(self):
        """Completely resets the scheduler instance from its config if state is corrupted."""
        try:
            self.pipe.scheduler = self.pipe.scheduler.from_config(self.pipe.scheduler.config)
            self._reset_scheduler()
            print("[TexDiff] Scheduler freshly re-instantiated from configuration.")
        except Exception as e:
            print(f"[TexDiff] Scheduler re-instantiation error: {e}")

    def sanitize_prompt(self, user_prompt: str) -> str:
        """
        Guarantees that the prompt strictly conditions the latent space into an orthogonal 2D CAD surface pattern.
        Keeps cultural/subject motifs at the front of the prompt (highest CLIP attention weight).
        """
        clean_p = user_prompt.strip()

        # If prompt already contains CAD specifications, keep cultural tokens in front
        if "2d flat textile" in clean_p.lower() or "seamless repeat tile" in clean_p.lower():
            final_p = clean_p
        else:
            final_p = f"{clean_p}, 2D flat textile CAD print, seamless repeat tile, orthogonal top-down flat lay view, continuous fabric print, crisp vector line contours, isolated texture"

        # Truncate overly long prompts to avoid CLIP 77-token overflow
        tokens = final_p.split()
        if len(tokens) > 72:
            final_p = " ".join(tokens[:72])
        return final_p

    def generate(
        self,
        prompt: str,
        negative_prompt: str = None,
        num_inference_steps: int = None,
        guidance_scale: float = 0.0,
        seed: int = 42,
        seamless: bool = True
    ) -> Image.Image:
        """
        Synthesizes a production-grade 512x512 seamless 2D CAD surface repeat tile.
        Self-healing: will NEVER crash or raise unhandled IndexError.
        """
        self.set_circular_padding(enable=seamless)
        self._reset_scheduler()

        engineered_prompt = self.sanitize_prompt(prompt)
        neg_prompt = negative_prompt if negative_prompt else self.default_negative_prompt

        # Default steps: 2 on CPU for rapid latency (~18s), 4 on GPU
        if num_inference_steps is None or num_inference_steps < 1:
            num_inference_steps = 4 if self.device == "cuda" else 2

        generator = torch.Generator(device=self.device).manual_seed(seed)

        # Primary attempt
        try:
            output = self.pipe(
                prompt=engineered_prompt,
                negative_prompt=neg_prompt,
                num_inference_steps=num_inference_steps,
                guidance_scale=guidance_scale,
                width=512,
                height=512,
                generator=generator
            )
            return output.images[0]
        except Exception as primary_err:
            print(f"[TexDiff] Primary generation notice ({primary_err}). Executing self-healing recovery...")
            # Recreate scheduler from scratch to purge corrupted step counters
            self._recreate_scheduler_fresh()

            try:
                # Retry with step=1 or step=2 on fresh scheduler
                recovery_steps = min(num_inference_steps, 2)
                generator_rec = torch.Generator(device=self.device).manual_seed(seed)
                output = self.pipe(
                    prompt=engineered_prompt,
                    negative_prompt=neg_prompt,
                    num_inference_steps=recovery_steps,
                    guidance_scale=guidance_scale,
                    width=512,
                    height=512,
                    generator=generator_rec
                )
                return output.images[0]
            except Exception as secondary_err:
                print(f"[TexDiff] Secondary recovery notice ({secondary_err}). Running CPU fail-safe generation...")
                self._recreate_scheduler_fresh()
                gen_safe = torch.Generator(device="cpu").manual_seed(seed)
                output = self.pipe(
                    prompt=engineered_prompt,
                    negative_prompt=neg_prompt,
                    num_inference_steps=1,
                    guidance_scale=guidance_scale,
                    width=512,
                    height=512,
                    generator=gen_safe
                )
                return output.images[0]

if __name__ == "__main__":
    tex_pipe = TexDiffPipeline()
    img = tex_pipe.generate("kanchipuram gold zari brocade", num_inference_steps=1, seed=42)
    os.makedirs("outputs", exist_ok=True)
    img.save("outputs/pipeline_test.png")
    print(f"Pipeline verification complete. Saved to outputs/pipeline_test.png (Size: {img.size})")

class FashionPromptParser:
    """
    Intelligent NLP Parser that extracts:
    - User's exact requested colors (multi-word and single-word)
    - User's exact requested attire/garment context
    - User's exact requested pattern styles and motifs
    - Explicit cultural heritage (if directly requested by user)
    """
    @staticmethod
    def parse(user_input: str) -> dict:
        raw = user_input.strip()
        low = f" {raw.lower()} "

        # 1. Extract Colors (Multi-word matches prioritized)
        found_colors = []
        found_hexes = []
        # Sort colors by length descending so "emerald green" matches before "green"
        for color_name in sorted(FASHION_COLORS.keys(), key=lambda x: -len(x)):
            pattern = r"(?<!\w)" + re.escape(color_name) + r"(?!\w)"
            if re.search(pattern, low):
                found_colors.append(color_name)
                found_hexes.append(FASHION_COLORS[color_name])
                low = re.sub(pattern, " ", low)

        # 2. Extract Attire Types
        found_attire = []
        attire_description = ""
        for attire_kw, desc in sorted(ATTIRE_TYPES.items(), key=lambda x: -len(x[0])):
            pattern = r"(?<!\w)" + re.escape(attire_kw) + r"(?!\w)"
            if re.search(pattern, low):
                found_attire.append(attire_kw)
                if not attire_description:
                    attire_description = desc
                low = re.sub(pattern, " ", low)

        # 3. Extract Design Patterns & Motifs
        found_styles = []
        style_descriptions = []
        for style_kw, desc in sorted(DESIGN_PATTERNS.items(), key=lambda x: -len(x[0])):
            pattern = r"(?<!\w)" + re.escape(style_kw) + r"(?!\w)"
            if re.search(pattern, low):
                found_styles.append(style_kw)
                style_descriptions.append(desc)
                low = re.sub(pattern, " ", low)

        # 4. Extract Specific Motifs (Peacock, Swan, Lotus, etc.)
        found_motifs = []
        motif_descriptions = []
        for motif_kw, desc in sorted(SPECIFIC_MOTIFS.items(), key=lambda x: -len(x[0])):
            pattern = r"(?<!\w)" + re.escape(motif_kw) + r"(?!\w)"
            if re.search(pattern, low):
                found_motifs.append(motif_kw)
                motif_descriptions.append(desc)
                low = re.sub(pattern, " ", low)

        # 5. Extract Cultural Heritage (ONLY when explicit triggers are present)
        matched_region = None
        for reg_id, data in CULTURAL_REGIONS.items():
            for trig in data["triggers"]:
                pattern = r"(?<!\w)" + re.escape(trig) + r"(?!\w)"
                if re.search(pattern, f" {raw.lower()} "):
                    matched_region = data
                    break
            if matched_region:
                break

        # Remaining residual keywords
        cleaned_residuals = re.sub(r"[^a-zA-Z0-9\s]", " ", low).split()
        stop_words = {
            "a", "an", "the", "in", "on", "of", "with", "and", "or", "for", "to",
            "pattern", "fabric", "textile", "print", "swatch", "tile", "seamless",
            "repeat", "surface", "design", "style", "texture", "make", "create",
            "draw", "generate", "render", "give", "me", "show", "material", "cloth"
        }
        residual_tokens = [w for w in cleaned_residuals if w not in stop_words and len(w) > 2]

        return {
            "raw_input": raw,
            "colors": found_colors,
            "hexes": found_hexes,
            "attire": found_attire,
            "attire_desc": attire_description,
            "styles": found_styles,
            "style_descs": style_descriptions,
            "motifs": found_motifs,
            "motif_descs": motif_descriptions,
            "cultural_region": matched_region,
            "residuals": residual_tokens
        }


class TextileMotifStore:
    """
    Universal Textile CAD Knowledge & Prompt Engineering Engine.
    Strictly prioritizes the user's requested colors, style, attire, and motifs,
    while guaranteeing pristine 2D orthogonal CAD repeat tiles with zero human figures or cloth folds.
    """
    def __init__(self, collection_name: str = "textile_motifs_v6"):
        self.client = chromadb.Client()
        self.encoder = SentenceTransformer("all-MiniLM-L6-v2")
        self.collection = self.client.get_or_create_collection(name=collection_name)
        self._seed_database()

    def _seed_database(self):
        if self.collection.count() >= len(CULTURAL_REGIONS):
            return
        ids, docs, metas = [], [], []
        for reg_id, data in CULTURAL_REGIONS.items():
            doc_text = f"{reg_id} {data['style']} {data['origin']} {data['technique']} {data['fabric']} {' '.join(data['triggers'])}"
            ids.append(f"reg_{reg_id}")
            docs.append(doc_text)
            metas.append({
                "region_id": reg_id,
                "style": data["style"],
                "origin": data["origin"],
                "technique": data["technique"],
                "fabric": data["fabric"],
                "palette": ", ".join(data["default_palette"])
            })
        embeddings = self.encoder.encode(docs).tolist()
        self.collection.add(ids=ids, documents=docs, metadatas=metas, embeddings=embeddings)

    def auto_expand_prompt(self, user_query: str, variation_index: int = None) -> dict:
        """
        Synthesizes a production CAD prompt where:
        1. User's exact colors, styles, attire, and motifs are placed in TOKENS 1-20 (maximum CLIP attention).
        2. Cultural weaving techniques are seamlessly incorporated if requested.
        3. Strict negative prompt eliminates all humans, garments, 3D folds, and wrinkles.
        4. Dynamic layout variation produces infinite unique generations for repeated inputs.
        """
        raw_query = user_query.strip() if user_query else "Royal Brocade"
        parsed = FashionPromptParser.parse(raw_query)

        # Select dynamic compositional layout
        if variation_index is not None:
            layout = COMPOSITION_LAYOUTS[variation_index % len(COMPOSITION_LAYOUTS)]
        else:
            layout = random.choice(COMPOSITION_LAYOUTS)

        # Strict Anti-Defect Negative Prompt
        strict_negative = (
            "human, person, woman, man, face, body, limbs, realistic model, photo of dress, "
            "3D perspective, folded fabric, wrinkles, cloth folds, ripples, crease, shadows, "
            "mannequin, garment mockup, depth of field, blurry edges, distorted lines, low quality"
        )

        # -------------------------------------------------------------
        # STEP 1: CONSTRUCT USER COLOR SPECIFICATION
        # -------------------------------------------------------------
        if parsed["colors"]:
            if len(parsed["colors"]) == 1:
                color_phrase = f"{parsed['colors'][0]}"
            elif len(parsed["colors"]) == 2:
                color_phrase = f"{parsed['colors'][0]} and {parsed['colors'][1]}"
            else:
                color_phrase = f"{', '.join(parsed['colors'][:-1])} and {parsed['colors'][-1]}"
            palette_hexes = parsed["hexes"]
            # Pad palette with neutral or accent if under 4 swatches
            while len(palette_hexes) < 4:
                palette_hexes.append("#D4AF37" if "#D4AF37" not in palette_hexes else "#111111")
        else:
            # If no colors specified by user, check cultural region or default
            if parsed["cultural_region"]:
                palette_hexes = parsed["cultural_region"]["default_palette"]
                color_phrase = ""
            else:
                color_phrase = "luxurious rich palette"
                palette_hexes = ["#D4AF37", "#800020", "#1B4D3E", "#0A192F"]

        # -------------------------------------------------------------
        # STEP 2: CONSTRUCT CORE DESIGN & MOTIF SPECIFICATION
        # -------------------------------------------------------------
        core_elements = []
        if color_phrase:
            core_elements.append(color_phrase)

        # Add explicit motifs (peacock, swan, lotus, etc.)
        if parsed["motif_descs"]:
            core_elements.append(parsed["motif_descs"][0])
        elif parsed["motifs"]:
            core_elements.append(f"{' and '.join(parsed['motifs'])} motifs")

        # Add design pattern/style (floral, houndstooth, polka dots, damask, etc.)
        if parsed["style_descs"]:
            core_elements.append(parsed["style_descs"][0])
        elif parsed["styles"]:
            core_elements.append(f"{' and '.join(parsed['styles'])} pattern")

        # If user didn't mention specific styles or motifs, use residual words or default
        if not parsed["motifs"] and not parsed["styles"]:
            if parsed["residuals"]:
                core_elements.append(f"{' '.join(parsed['residuals'])} pattern")
            elif not parsed["cultural_region"]:
                core_elements.append("intricate damask filigree pattern")

        core_subject = ", ".join(core_elements)

        # -------------------------------------------------------------
        # STEP 3: CONSTRUCT ATTIRE & FABRIC CONTEXT
        # -------------------------------------------------------------
        if parsed["attire_desc"]:
            attire_context = f"for {parsed['attire_desc']}"
        elif parsed["attire"]:
            attire_context = f"for bespoke {parsed['attire'][0]}"
        else:
            attire_context = "for bespoke haute couture fashion"

        # -------------------------------------------------------------
        # STEP 4: WEAVE IN CULTURAL TECHNIQUE (IF APPLICABLE)
        # -------------------------------------------------------------
        if parsed["cultural_region"]:
            reg = parsed["cultural_region"]
            matched_style = reg["style"]
            origin = reg["origin"]
            technique = reg["technique"]
            fabric = reg["fabric"]
            cultural_clause = f", authentic {reg['style']} {technique}, woven on {fabric}"
        else:
            matched_style = f"Haute Couture ({raw_query.title()})"
            origin = "Custom Atelier Studio"
            cultural_clause = ", fine textile weave texture"

        # -------------------------------------------------------------
        # STEP 5: ASSEMBLE 2D FLAT CAD SURFACE PROMPT
        # -------------------------------------------------------------
        # Tokens 1-25 contain the user's EXACT colors, motifs, style, and attire!
        engineered_prompt = (
            f"2D flat CAD textile print, seamless repeat tile, {core_subject} {attire_context}"
            f"{cultural_clause}, {layout}, orthogonal top-down flat lay view, "
            f"continuous surface pattern, crisp vector line contours, production textile swatch"
        )

        # Compact prompt to strictly under 70 tokens for 100% CLIP coverage
        tokens = engineered_prompt.split()
        if len(tokens) > 68:
            engineered_prompt = " ".join(tokens[:68])

        return {
            "matched_style": matched_style,
            "origin": origin,
            "engineered_prompt": engineered_prompt,
            "negative_prompt": strict_negative,
            "palette": ", ".join(palette_hexes[:4]),
            "extracted_colors": parsed["colors"],
            "extracted_attire": parsed["attire"],
            "extracted_styles": parsed["styles"]
        }

def image_to_base64_data_url(image: Image.Image) -> str:
    buffered = io.BytesIO()
    image.save(buffered, format="PNG")
    img_b64 = base64.b64encode(buffered.getvalue()).decode("utf-8")
    return f"data:image/png;base64,{img_b64}"


def create_tiled_canvas(tile: Image.Image, repeat_count: int = 3) -> Image.Image:
    w, h = tile.size
    canvas = Image.new("RGB", (w * repeat_count, h * repeat_count))
    for i in range(repeat_count):
        for j in range(repeat_count):
            canvas.paste(tile, (i * w, j * h))
    return canvas


def extract_palette(image: Image.Image, num_colors: int = 4):
    img = image.resize((64, 64)).convert("RGB")
    result = img.quantize(colors=num_colors, method=Image.Quantize.FASTOCTREE)
    pal = result.getpalette()[:num_colors * 3]
    return [f"#{pal[i]:02x}{pal[i+1]:02x}{pal[i+2]:02x}".upper() for i in range(0, len(pal), 3)]

﻿import os
import torch
from PIL import Image
from transformers import CLIPProcessor, CLIPModel
from vector_store import TextileMotifStore

class TexDiffEvaluator:
    def __init__(self, clip_model_id: str = "openai/clip-vit-base-patch32"):
        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        self.model = CLIPModel.from_pretrained(clip_model_id).to(self.device)
        self.processor = CLIPProcessor.from_pretrained(clip_model_id)
        self.motif_store = TextileMotifStore()

    def evaluate_retrieval(self):
        tests = [
            ("hand printed lotus vines madder red", "Andhra Kalamkari"),
            ("kanchipuram gold zari border temple rekku", "Kanchipuram Zari Brocade"),
            ("madurai sungudi tie dye ring dots", "Madurai Sungudi")
        ]
        hits = 0
        for q, expected in tests:
            res = self.motif_store.auto_expand_prompt(q)
            if res["matched_style"].lower() == expected.lower():
                hits += 1
        return hits / len(tests)
